My Java Project

Java Number Guessing Game

This is my Java Console-Based Number Guessing Game. The player must guess the secret number from 1 to 100 within 7 attempts. The game gives hints whether the guess is too high or too low and calculates the player's score. 

import java.util.Scanner;
import java.util.Random;

public class NumberGuessingGame {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        Random random = new Random();

        int playAgain = 1;

        System.out.println("================================");
        System.out.println("      NUMBER GUESSING GAME");
        System.out.println("================================");
        System.out.println("Guess the secret number from 1 to 100.");
        System.out.println("You only have 7 attempts!");

        while (playAgain == 1) {

            int secretNumber = random.nextInt(100) + 1;
            int attempts = 0;
            int maxAttempts = 7;
            boolean guessedCorrectly = false;

            System.out.println("\nA new number has been generated!");

            while (attempts < maxAttempts) {

                System.out.print("\nEnter your guess: ");

                while (!input.hasNextInt()) {
                    System.out.print("Invalid input! Enter a number: ");
                    input.next();
                }

                int guess = input.nextInt();

                if (guess < 1 || guess > 100) {
                    System.out.println("Please enter a number from 1 to 100.");
                    continue;
                }

                attempts++;

                if (guess == secretNumber) {
                    System.out.println("Congratulations! You guessed correctly!");
                    guessedCorrectly = true;
                    break;
                }
                else if (guess < secretNumber) {
                    System.out.println("Too LOW! Try a higher number.");
                }
                else {
                    System.out.println("Too HIGH! Try a lower number.");
                }

                System.out.println("Attempts remaining: "
                        + (maxAttempts - attempts));
            }

            // Display game results
            System.out.println("\n========== GAME RESULT ==========");

            if (guessedCorrectly) {

                int score = (maxAttempts - attempts + 1) * 10;

                System.out.println("Secret Number: " + secretNumber);
                System.out.println("Attempts Used: " + attempts);
                System.out.println("Your Score: " + score);
                System.out.println("Result: YOU WIN!");

            } else {

                System.out.println("Secret Number: " + secretNumber);
                System.out.println("Attempts Used: " + attempts);
                System.out.println("Your Score: 0");
                System.out.println("Result: GAME OVER!");

            }

            System.out.println("=================================");

            System.out.println("\nDo you want to play again?");
            System.out.println("1 - YES");
            System.out.println("2 - NO");
            System.out.print("Enter your choice: ");

            while (!input.hasNextInt()) {
                System.out.print("Invalid choice! Enter 1 or 2: ");
                input.next();
            }

            playAgain = input.nextInt();

            while (playAgain != 1 && playAgain != 2) {
                System.out.print("Please enter only 1 or 2: ");

                while (!input.hasNextInt()) {
                    System.out.print("Invalid input! Enter 1 or 2: ");
                    input.next();
                }

                playAgain = input.nextInt();
            }
        }

        System.out.println("\nThank you for playing!");
        System.out.println("GAME EXITED.");

        input.close();
    }
}
